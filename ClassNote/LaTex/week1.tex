% ══ 4. VACUOUS TRUTH ══════════════════════════════════════
\begin{frame}[t]{Vacuous Truth}
    \footnotesize
    \begin{block}{정의}
        $p \rightarrow q$ 에서 가정 $p$가 거짓(F)일 때, 결론 $q$에 무관하게 전체 명제가 참(T)이 되는 현상.
    \end{block}

    \vspace{0.08cm}
    \textbf{증명 1 — 논리적 동치 (Equivalence) 로 증명}
    \begin{exampleblock}{}
        \scriptsize
        $p \rightarrow q \;\equiv\; \neg p \lor q$
        \qquad
        $p = F \;\Rightarrow\; \neg p = T \;\Rightarrow\; T \lor q = \mathbf{T}$ \quad ($q$ 값과 완전히 무관)
    \end{exampleblock}

    \vspace{0.1cm}
    \textbf{증명 2 — 반례 (Counterexample) 의 부재}
    \begin{block}{}
        \scriptsize
        $p \rightarrow q$ 가 거짓이 되는 유일한 경우: \textbf{``$p = T$ 인데 $q = F$''} 뿐입니다.\\
        $p$가 거짓이라면 이 반례를 구성할 수단 자체가 없으므로, 수학적으로 참으로 인정합니다.
    \end{block}

    \vspace{0.1cm}
    \begin{columns}[T]
        \column{0.45\textwidth}
        \begin{exampleblock}{공허한 참 ($F \rightarrow T$)}
            \centering $F \rightarrow T \;=\; \mathbf{T}$ \; $\checkmark$
        \end{exampleblock}
        \column{0.45\textwidth}
        \begin{alertblock}{유일한 거짓 ($T \rightarrow F$)}
            \centering $T \rightarrow F \;=\; \mathbf{F}$ \; $\times$
        \end{alertblock}
    \end{columns}
\end{frame}

% ══ 5. VACUOUS TRUTH IN C++ ═══════════════════════════════
\begin{frame}[fragile,t]{Vacuous Truth (Cont.)}
    \footnotesize
    \begin{block}{한정자와 빈 집합}
        $\forall x \in \emptyset,\; P(x)$ 는 명백한 공허한 참 — 반례를 찾을 수단 자체가 없으므로 항상 T
    \end{block}

\begin{lstlisting}[style=cpp]
#include <algorithm>
#include <vector>
std::vector<int> empty_vec;   // empty array (no elements)
// "Every element of this array is even."
// No counterexample can be found -> Vacuously True
bool result = std::all_of(
    empty_vec.begin(), empty_vec.end(),
    [](int x) { return x % 2 == 0; }
);
// result == true  <- Vacuous Truth in action
\end{lstlisting}

    \begin{alertblock}{REMARK}
        \scriptsize
        빈 배열이나 Null 상태의 데이터가 입력됐을 때, Validation 루프가 공허한 참으로 인해
        \texttt{true}를 반환하여 보안·접근 제어가 \textbf{뚫리는 버그}가 자주 발생합니다.
    \end{alertblock}
\end{frame}

% ══ 8. LOGICAL EQUIVALENCES ═══════════════════════════════
\begin{frame}[t]{Logical Equivalences}
    \footnotesize
    복잡한 명제(조건문)를 수학적으로 \textbf{Refactoring}하기 위한 핵심 법칙들입니다.
    \vspace{0.08cm}

    \begin{center}
    \renewcommand{\arraystretch}{1.65}
    \small
    \begin{tabular}{ll}
        \toprule
        \textbf{동치 법칙 (Equivalence)} & \textbf{이름 (Name)} \\
        \midrule
        $p \land \mathbf{T} \equiv p \quad / \quad p \lor \mathbf{F} \equiv p$ & Identity laws \\
        $p \lor \mathbf{T} \equiv \mathbf{T} \quad / \quad p \land \mathbf{F} \equiv \mathbf{F}$ & Domination laws \\
        $p \lor p \equiv p \quad / \quad p \land p \equiv p$ & Idempotent laws \\
        $\neg(\neg p) \equiv p$ & Double negation law \\
        $p \lor (q \land r) \equiv (p \lor q) \land (p \lor r)$ & Distributive laws \\
        $\neg(p \land q) \equiv \neg p \lor \neg q$ & De Morgan's laws \\
        $p \lor (p \land q) \equiv p \quad / \quad p \land (p \lor q) \equiv p$ & Absorption laws \\
        \midrule
        \rowcolor{lightblue}
        $p \rightarrow q \equiv \neg p \lor q$ & \textbf{Conditional $\bigstar$ 핵심} \\
        \bottomrule
    \end{tabular}
    \renewcommand{\arraystretch}{1}
    \end{center}
\end{frame}

% ══ 9. SPEC → PROPOSITION ═════════════════════════════════
\begin{frame}[fragile,t]{Proposition으로 변환}
    \footnotesize
    \begin{block}{요구사항 \small(변수: $A$=Admin, $C$=Active, $P$=Premium)}
        ``유저가 관리자이면서 활성 상태이거나, 관리자는 아니지만 활성 상태이고 프리미엄이거나,
        활성 상태이면서 프리미엄이 아닌 경우에 한하여 접근을 허용하라.''
    \end{block}

    \textbf{문장 $\rightarrow$ 기호 대응:}
    \begin{center}
    \scriptsize
    \renewcommand{\arraystretch}{1.0}
    \begin{tabular}{ll}
        ``관리자이면서 활성'' & $\Rightarrow\; A \land C$ \\
        ``관리자는 아니지만 활성 + 프리미엄'' & $\Rightarrow\; \neg A \land C \land P$ \\
        ``활성이면서 프리미엄이 아님'' & $\Rightarrow\; C \land \neg P$ \\
        세 조건 중 하나라도 만족 & $\Rightarrow\; \lor$ 로 연결 \\
    \end{tabular}
    \renewcommand{\arraystretch}{1}
    \end{center}

    \textbf{직역 Proposition:}
    \[
        (A \land C) \;\lor\; (\neg A \land C \land P) \;\lor\; (C \land \neg P)
    \]

\begin{lstlisting}[style=cpp]
if ((isAdmin && isActive) ||
    (!isAdmin && isActive && isPremium) ||
    (isActive && !isPremium)) { grantAccess(); }
\end{lstlisting}
    \begin{exampleblock}{$\Rightarrow$ 질문}
        \footnotesize Logical Equivalences를 사용하면 이 식을 더 간단하게 줄일 수 있을까요?
    \end{exampleblock}
\end{frame}

% ══ 10. PROOF ═════════════════════════════════════════════
\begin{frame}[t]{Proposition Reduction}
    \footnotesize
    \textbf{주어진 명제:} $(A \land C) \lor (\neg A \land C \land P) \lor (C \land \neg P)$

    \setlength{\abovedisplayskip}{3pt}
    \setlength{\belowdisplayskip}{3pt}
    {\setlength{\jot}{3pt}%
    \begin{align*}
        & \equiv C \land \bigl(A \lor (\neg A \land P) \lor \neg P\bigr)
            && \text{\scriptsize Distributive law ($C$로 묶기)} \\
        & \equiv C \land (A \lor P \lor \neg P)
            && \text{\scriptsize Absorption: $X \lor (\neg X \land Y) \equiv X \lor Y$} \\
        & \equiv C \land (A \lor \mathbf{T})
            && \text{\scriptsize Negation: $P \lor \neg P \equiv \mathbf{T}$} \\
        & \equiv C \land \mathbf{T}
            && \text{\scriptsize Domination: $A \lor \mathbf{T} \equiv \mathbf{T}$} \\
        & \equiv \boxed{C}
            && \text{\scriptsize Identity: $C \land \mathbf{T} \equiv C$ \quad $\checkmark$ 완성!}
    \end{align*}}

    \vspace{0.04cm}
    \begin{exampleblock}{증명 결과}
        복잡했던 기획서의 진짜 의도는 단 하나였습니다.\\
        \textbf{``활성 상태($C$)인 유저에게 접근을 허용하라.''}
    \end{exampleblock}
\end{frame}

% ══ 12. PREDICATE ═════════════════════════════════════════
\begin{frame}[fragile,t]{Predicate \& Quantifier}
    \footnotesize
    \begin{alertblock}{REMARK}
        \scriptsize 지금까지는 \textbf{고정된 값}의 참/거짓을 다뤘습니다. 실제 코드는 \textbf{변수}를 다룹니다. $\rightarrow$ Predicate(술어)가 필요합니다.
    \end{alertblock}

    \vspace{0.05cm}
    \begin{block}{Predicate (술어)란?}
        \footnotesize 주어의 성질이나 관계를 나타내는 말. 명제 함수 $P(x)$는 변수가 채워져야 참/거짓 판별 가능.
    \end{block}

    \vspace{0.05cm}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \textbf{$\forall x\, P(x)$ — Universal}\\
        \scriptsize 모든 $x$에 대해 참. 반례 하나라도 있으면 거짓.
\begin{lstlisting}[style=cpp]
// All elements even?
std::all_of(v.begin(), v.end(),
  [](int x){ return x%2==0; });
// empty vec -> Vacuously True!
\end{lstlisting}

        \column{0.48\textwidth}
        \textbf{$\exists x\, P(x)$ — Existential}\\
        \scriptsize 조건을 만족하는 $x$가 하나라도 존재.
\begin{lstlisting}[style=cpp]
// Any even element?
std::any_of(v.begin(), v.end(),
  [](int x){ return x%2==0; });
// empty vec -> always false
\end{lstlisting}
    \end{columns}

    \vspace{0.04cm}
    \begin{block}{부정 (Negation)}
        \footnotesize
        $\neg(\forall x\, P(x)) \equiv \exists x\, \neg P(x)$
        \hfill
        $\neg(\exists x\, P(x)) \equiv \forall x\, \neg P(x)$
    \end{block}
\end{frame}

% ══ 13. SAT ═══════════════════════════════════════════════
\begin{frame}[t]{Satisfiability (SAT)}
    \footnotesize
    \begin{block}{Satisfiability 란?}
        주어진 복합 명제가 \textbf{참(True)}이 되게 만드는 변수값 조합(Truth Assignment)이
        \textbf{적어도 하나 존재}하는가?
    \end{block}

    \vspace{0.08cm}
    \begin{columns}[T]
        \column{0.32\textwidth}
        \begin{exampleblock}{Satisfiable}
            \scriptsize $p \lor \neg q$\\[3pt]
            참이 되는 조합 존재.\\
            예: $p{=}T$ 이면 참.
        \end{exampleblock}
        \column{0.32\textwidth}
        \begin{alertblock}{Contradiction}
            \scriptsize $p \land \neg p$\\[3pt]
            참이 되는 조합 없음.\\
            = Unsatisfiable
        \end{alertblock}
        \column{0.32\textwidth}
        \begin{block}{Tautology}
            \scriptsize $p \lor \neg p$\\[3pt]
            항상 참.\\
            Satisfiable의 특수 경우
        \end{block}
    \end{columns}

    \vspace{0.15cm}
    \begin{exampleblock}{REMARK}
        \scriptsize
        현실의 복잡한 \textbf{제약 조건 문제}(배치, 스케줄링, 회로 설계 등)를
        명제 논리식으로 변환하면,\\
        그 식이 \textbf{Satisfiable 한지} 판단하는 것 = 제약 조건을 만족하는 \textbf{해(Solution)를 찾는 것}입니다.\\[3pt]
        \textbf{$\rightarrow$ 지금부터 n-Queens 문제를 SAT 로 직접 모델링해봅시다!}
    \end{exampleblock}
\end{frame}

% ══ 14. n-QUEENS 문제 정의 ════════════════════════════════
\begin{frame}[t]{n-Queens}
    \footnotesize
    \begin{block}{문제 (Problem Statement)}
        $n \times n$ 체스판에 $n$개의 퀸(Queen)을 배치하되,
        \textbf{어떤 두 퀸도 서로 공격하지 못하게} 할 수 있는가?\\
        \scriptsize (퀸은 같은 행·열·대각선 위의 모든 칸을 공격합니다.)
    \end{block}

    \vspace{0.12cm}
    \begin{columns}[T]
        \column{0.38\textwidth}
        \textbf{4-Queens 해답 예시}
        \vspace{0.08cm}

        \centering
        \setlength{\tabcolsep}{4pt}
        \renewcommand{\arraystretch}{1.4}
        \scriptsize
        \begin{tabular}{|c|c|c|c|}
            \hline
            \cellcolor{gray!20} & \Qcell & \cellcolor{gray!20} & \cellcolor{white} \\
            \hline
            \cellcolor{white} & \cellcolor{gray!20} & \cellcolor{white} & \Qcell \\
            \hline
            \Qcell & \cellcolor{white} & \cellcolor{gray!20} & \cellcolor{white} \\
            \hline
            \cellcolor{gray!20} & \cellcolor{white} & \Qcell & \cellcolor{gray!20} \\
            \hline
        \end{tabular}
        \renewcommand{\arraystretch}{1}

        \vspace{0.08cm}
        \scriptsize\centering 행·열·대각선 모두 충돌 없음

        \column{0.58\textwidth}
        \textbf{SAT 변수 정의}

        \begin{exampleblock}{}
            \footnotesize
            $p(i,\, j)$ = ``$i$행 $j$열에 퀸이 있다''\\[3pt]
            \scriptsize
            \textbf{True}: 해당 칸에 퀸 있음\\
            \textbf{False}: 해당 칸에 퀸 없음\\[4pt]
            $n \times n$ 개의 불리언 변수 $p(1,1),\, p(1,2),\, \ldots,\, p(n,n)$
        \end{exampleblock}

        \begin{block}{우리의 목표}
            \footnotesize 다음 \textbf{5가지 제약 조건}을 모두 만족하는\\
            $p(i,j)$ 값의 조합을 찾아라!\\[2pt]
            \scriptsize $\Rightarrow$ 이것이 바로 SAT 문제입니다.
        \end{block}
    \end{columns}
\end{frame}

% ══ 15. Q1 ════════════════════════════════════════════════
\begin{frame}[t]{n-Queens 모델링: $Q_1$}
    \footnotesize

    \begin{exampleblock}{조건 $Q_1$: 각 행(Row)에는 퀸이 \textbf{최소 1개} 있어야 한다}
        \[
            Q_1 = \bigwedge_{i=1}^{n} \Bigl(\bigvee_{j=1}^{n} p(i,\, j)\Bigr)
        \]
        \scriptsize REMARK: $\bigwedge$ = 큰 AND, $\bigvee$ = 큰 OR — 여러 항을 한꺼번에 묶는 표기법입니다.
    \end{exampleblock}

    \vspace{0.06cm}
    \textbf{직관적으로 이해하기:}
    \begin{itemize}
        \setlength\itemsep{0.06em}
        \item $\bigvee_{j=1}^{n} p(i,j)$\,:
              ``$i$번 행의 어떤 열에든 \textbf{하나라도} 퀸이 있다''\\
              $= p(i,1) \lor p(i,2) \lor \cdots \lor p(i,n)$
        \item $\bigwedge_{i=1}^{n}$\,:
              위 조건을 \textbf{모든 행(1행부터 $n$행까지)}에 동시에 요구한다
    \end{itemize}

    \vspace{0.05cm}
    \begin{block}{4-Queens 예시 ($n = 4$)}
        \tiny
        $Q_1 = \bigl(p(1,1) \lor p(1,2) \lor p(1,3) \lor p(1,4)\bigr)$\\
        $\phantom{Q_1 = }\land \bigl(p(2,1) \lor p(2,2) \lor p(2,3) \lor p(2,4)\bigr)$\\
        $\phantom{Q_1 = }\land \bigl(p(3,1) \lor p(3,2) \lor p(3,3) \lor p(3,4)\bigr)$\\
        $\phantom{Q_1 = }\land \bigl(p(4,1) \lor p(4,2) \lor p(4,3) \lor p(4,4)\bigr)$
    \end{block}
\end{frame}

% ══ 16. Q2 ════════════════════════════════════════════════
\begin{frame}[t]{n-Queens 모델링: $Q_2$}
    \footnotesize

    \begin{alertblock}{조건 $Q_2$: 각 행(Row)에는 퀸이 \textbf{최대 1개}여야 한다}
        \[
            Q_2 = \bigwedge_{i=1}^{n}\; \bigwedge_{j=1}^{n-1}\; \bigwedge_{k=j+1}^{n}
            \bigl(\neg p(i,j) \lor \neg p(i,k)\bigr)
        \]
    \end{alertblock}

    \vspace{0.08cm}
    \textbf{왜 이 식이 ``최대 1개''를 표현할까? — 단계별 논리 전개}

    \begin{block}{}
        \scriptsize
        \begin{enumerate}\setlength\itemsep{0.25em}
            \item \textbf{목표}: 같은 행 $i$에서, 임의의 두 열 $j \neq k$에 퀸이 \textbf{동시에 있으면 안 됨}
            \item \textbf{금지 조건}: $p(i,j) \land p(i,k) = \mathbf{False}$
            \item \textbf{양쪽에 $\neg$ 를 씌우기}:
                  $\neg\bigl(p(i,j) \land p(i,k)\bigr) = \mathbf{True}$
            \item \textbf{De Morgan 적용}:
                  $\neg p(i,j) \lor \neg p(i,k) = \mathbf{True}$\\
                  ``두 칸 중 \textbf{적어도 하나는 반드시 비어 있어야} 한다''
            \item \textbf{모든 쌍·모든 행에 적용}: $j{<}k$인 모든 쌍, 모든 행 $i$에 대해 $\bigwedge$ 으로 묶기
        \end{enumerate}
    \end{block}

    \vspace{0.04cm}
    \begin{exampleblock}{}
        \centering\footnotesize
        $Q_1 \land Q_2$ = 각 행에 \textbf{정확히 1개}의 퀸이 존재
    \end{exampleblock}
\end{frame}

% ══ 17. Q3 ════════════════════════════════════════════════
\begin{frame}[t]{n-Queens 모델링: $Q_3$}
    \footnotesize

    \begin{alertblock}{조건 $Q_3$: 각 열(Column)에는 퀸이 \textbf{최대 1개}여야 한다}
        \[
            Q_3 = \bigwedge_{j=1}^{n}\; \bigwedge_{i=1}^{n-1}\; \bigwedge_{k=i+1}^{n}
            \bigl(\neg p(i,j) \lor \neg p(k,j)\bigr)
        \]
    \end{alertblock}

    \vspace{0.04cm}
    \textbf{$Q_2$와 완전히 같은 논리 — 행과 열만 바뀐 것!}

    \begin{block}{$Q_2$와 대응 비교}
        \tiny
        \renewcommand{\arraystretch}{1.1}
        \begin{tabular}{lll}
            \toprule
            & \textbf{$Q_2$ (행 제약)} & \textbf{$Q_3$ (열 제약)} \\
            \midrule
            고정 & 행 $i$ 고정 & 열 $j$ 고정 \\
            비교 & 열 $j \neq k$ 비교 & 행 $i \neq k$ 비교 \\
            금지 & $\neg p(i,j) \lor \neg p(i,k)$ & $\neg p(i,j) \lor \neg p(k,j)$ \\
            의미 & 같은 행에 두 퀸 금지 & 같은 열에 두 퀸 금지 \\
            \bottomrule
        \end{tabular}
        \renewcommand{\arraystretch}{1}
    \end{block}

    \vspace{0.02cm}
    \begin{exampleblock}{직관 + 요약}
        \scriptsize
        같은 열에서 서로 다른 두 행을 뽑으면 둘 다 퀸이면 안 됩니다.
        $\rightarrow$ De Morgan: $\neg p(i,j) \lor \neg p(k,j)$\\[2pt]
        \centering\textbf{$Q_1 \land Q_2 \land Q_3$ 완료 $\Rightarrow$ 행·열 제약 달성. 남은 것: 대각선!}
    \end{exampleblock}
\end{frame}

% ══ 18. Q4 / Q5 ═══════════════════════════════════════════
\begin{frame}{n-Queens 모델링: $Q_4,\, Q_5$}
    \scriptsize

    \begin{columns}[T]
        \column{0.52\textwidth}
        \begin{alertblock}{$Q_4$: $\nearrow$ 방향 대각선 — 최대 1개}
            \small$Q_4 = \bigwedge_{i,j,k \geq 1} \bigl(\neg p(i,j) \lor \neg p(i{-}k,\, j{+}k)\bigr)$\\[1pt]
            \footnotesize
            $(i,j)$에서 \textbf{위·오른쪽}으로 $k$칸 $\Rightarrow$ 같은 $\nearrow$ 대각선\\
            두 칸에 동시에 퀸 금지 $\rightarrow$ De Morgan 적용
        \end{alertblock}

        \vspace{0.04cm}
        \begin{alertblock}{$Q_5$: $\searrow$ 방향 대각선 — 최대 1개}
            \small$Q_5 = \bigwedge_{i,j,k \geq 1} \bigl(\neg p(i,j) \lor \neg p(i{+}k,\, j{+}k)\bigr)$\\[1pt]
            \footnotesize
            $(i,j)$에서 \textbf{아래·오른쪽}으로 $k$칸 $\Rightarrow$ 같은 $\searrow$ 대각선\\
            두 칸에 동시에 퀸 금지 $\rightarrow$ De Morgan 적용
        \end{alertblock}

        \vspace{0.04cm}
        \begin{exampleblock}{최종 SAT Equation}
            \centering
            \small$Q = Q_1 \land Q_2 \land Q_3 \land Q_4 \land Q_5$\\[2pt]
            \footnotesize 이 식을 \textbf{True}로 만드는 $p(i,j)$ 조합 $=$ $n$-Queens 정답
        \end{exampleblock}

        \column{0.44\textwidth}
   
        \footnotesize 같은 색 = 같은 $\searrow$ 대각선
        \vspace{0.04cm}

        \centering
        \setlength{\tabcolsep}{1pt}
        \renewcommand{\arraystretch}{1.3}
        \begin{tabular}{|c|c|c|c|}
            \hline
            \cellcolor{lightblue!80}\tiny(1,1) & \cellcolor{lightgreen!60}\tiny(1,2) & \cellcolor{lightamber!80}\tiny(1,3) & \cellcolor{lightred!60}\tiny(1,4)\\
            \hline
            \cellcolor{accentblue!25}\tiny(2,1) & \cellcolor{lightblue!80}\tiny(2,2) & \cellcolor{lightgreen!60}\tiny(2,3) & \cellcolor{lightamber!80}\tiny(2,4)\\
            \hline
            \cellcolor{navyblue!15}\tiny(3,1) & \cellcolor{accentblue!25}\tiny(3,2) & \cellcolor{lightblue!80}\tiny(3,3) & \cellcolor{lightgreen!60}\tiny(3,4)\\
            \hline
            \cellcolor{gray!25}\tiny(4,1) & \cellcolor{navyblue!15}\tiny(4,2) & \cellcolor{accentblue!25}\tiny(4,3) & \cellcolor{lightblue!80}\tiny(4,4)\\
            \hline
        \end{tabular}
        \renewcommand{\arraystretch}{1}

        \vspace{0.08cm}
        $(i,j)$와 $(i',j')$가 같은 $\searrow$ 대각선\\
        $\Leftrightarrow\; i' - i = j' - j$\\
        $\Leftrightarrow\; (i',j') = (i{+}k,\, j{+}k)$\\[4pt]
        $\nearrow$ 대각선은?\\
        $\Leftrightarrow\; i' - i = -(j' - j)$\\
        $\Leftrightarrow\; (i',j') = (i{-}k,\, j{+}k)$
    \end{columns}
\end{frame}
